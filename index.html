    <script>

        // 初始化預設資料庫 (儲存在記憶體中，可延伸 LocalStorage)

        let financeItems = [

            { id: 1, title: '基本薪資', type: 'income', freq: 'monthly', amount: 50000 },

            { id: 2, title: '日常伙食', type: 'expense', freq: 'daily', amount: 300 },

            { id: 3, title: '房租房貸', type: 'expense', freq: 'monthly', amount: 12000 }

        ];


        // 設定預設目標日期為一年後

        window.onload = function() {

            const today = new Date();

            today.setFullYear(today.getFullYear() + 1);

            document.getElementById('targetDate').value = today.toISOString().split('T')[0];

            

            renderList();

            calculateFinances();

        };


        // 渲染收支明細列表

        function renderList() {

            const listEl = document.getElementById('flowList');

            listEl.innerHTML = '';

            

            const freqText = { daily: '每日', weekly: '每週', monthly: '每月', yearly: '每年' };


            financeItems.forEach(item => {

                const li = document.createElement('li');

                li.className = 'item-todo';

                

                const isIncome = item.type === 'income';

                const amtClass = isIncome ? 'amt-income' : 'amt-expense';

                const prefix = isIncome ? '+' : '-';


                li.innerHTML = `

                    <div class="item-info">

                        <div class="title">${item.title}</div>

                        <div class="freq">${freqText[item.freq]}</div>

                    </div>

                    <div>

                        <span class="${amtClass}">${prefix}$${item.amount}</span>

                        <button class="delete-btn" onclick="deleteItem(${item.id})">❌</button>

                    </div>

                `;

                listEl.appendChild(li);

            });

        }


        // 新增收支項目

        function addFlowItem() {

            const titleEl = document.getElementById('flowTitle');

            const typeEl = document.getElementById('flowType');

            const freqEl = document.getElementById('flowFreq');

            const amountEl = document.getElementById('flowAmount');


            if (!titleEl.value.trim() || !amountEl.value || amountEl.value <= 0) {

                alert('請填寫正確的項目名稱與金額');

                return;

            }


            const newItem = {

                id: Date.now(),

                title: titleEl.value.trim(),

                type: typeEl.value,

                freq: freqEl.value,

                amount: parseFloat(amountEl.value)

            };


            financeItems.push(newItem);

            

            // 清空輸入欄位

            titleEl.value = '';

            amountEl.value = '';


            renderList();

            calculateFinances();

        }


        // 刪除收支項目

        function deleteItem(id) {

            financeItems = financeItems.filter(item => item.id !== id);

            renderList();

            calculateFinances();

        }


        // 核心財務邏輯計算

        function calculateFinances() {

            const targetAmount = parseFloat(document.getElementById('targetAmount').value) || 0;

            

            let totalWeeklyNet = 0;


            financeItems.forEach(item => {

                let weeklyEquivalent = 0;

                

                // 將各個時間頻率標準化轉化為「每週」基礎

                // 依據標準：1年 = 52週, 1個月 = 4.333週, 1天 = 7天/週

                switch (item.freq) {

                    case 'daily':

                        weeklyEquivalent = item.amount * 7;

                        break;

                    case 'weekly':

                        weeklyEquivalent = item.amount;

                        break;

                    case 'monthly':

                        weeklyEquivalent = item.amount / 4.3333;

                        break;

                    case 'yearly':

                        weeklyEquivalent = item.amount / 52;

                        break;

                }


                if (item.type === 'income') {

                    totalWeeklyNet += weeklyEquivalent;

                } else {

                    totalWeeklyNet -= weeklyEquivalent;

                }

            });


            // 渲染淨儲蓄顯示

            const monthlyNet = totalWeeklyNet * 4.3333;

            document.getElementById('weeklyNetSavings').innerText = `$${Math.round(totalWeeklyNet).toLocaleString()}`;

            document.getElementById('monthlyNetSavings').innerText = `$${Math.round(monthlyNet).toLocaleString()}`;


            const weeksValueEl = document.getElementById('weeksToTarget');

            const dateValueEl = document.getElementById('estimatedDate');


            // 處理儲蓄防呆機制（若小於等於 0 則無法算得未來達成日）

            if (totalWeeklyNet <= 0 || targetAmount <= 0) {

                weeksValueEl.innerText = '永無止境';

                dateValueEl.innerText = '請增加收入或降低支出';

                return;

            }


            // 計算所需週數與目標日期

            const weeksNeeded = targetAmount / totalWeeklyNet;

            weeksValueEl.innerText = `${Math.ceil(weeksNeeded)} 週`;


            const today = new Date();

            const daysNeeded = weeksNeeded * 7;

            const estimatedDate = new Date(today.getTime() + daysNeeded * 24 * 60 * 60 * 1000);

            

            // 格式化日期輸出 (YYYY-MM-DD)

            const year = estimatedDate.getFullYear();

            const month = String(estimatedDate.getMonth() + 1).padStart(2, '0');

            const day = String(estimatedDate.getDate()).padStart(2, '0');

            

            dateValueEl.innerText = `${year}-${month}-${day}`;

        }
